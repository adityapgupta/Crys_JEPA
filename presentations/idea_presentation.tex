\begin{frame}{Motivation}
\begin{itemize}\setlength\itemsep{4mm}
\item Diffusion models generate many candidate crystals.
\item Crys-JEPA ranks candidates by their similarity to known reference crystals.
\item The reference choice controls which candidates enter the expensive refinement loop.
\item A useful reference set should capture structural context and improve the number of stable candidates selected at a fixed budget.
\end{itemize}
\vspace{4mm}

\begin{frame}{Original reference score}
For a candidate $c$, collect known references whose elements are contained in the candidate system:
\[
R(c)=\{r\in\mathcal D_{\mathrm{MP20}}:\operatorname{elements}(r)\subseteq\operatorname{elements}(c)\}.
\]
The original score is
\[
s(c)=\frac{1}{|R(c)|}\sum_{r\in R(c)}\lVert h_c-h_r\rVert_2^2.
\]
\begin{itemize}\setlength\itemsep{3mm}
\item Smaller scores rank first.
\item Every reference structure receives equal weight.
\item Different polymorphs of one formula remain separate structures.
\item We define: A chemical system is a set of elements. A reference subsystem is a known crystal whose elements are a subset of that system.
\end{itemize}
\end{frame}

\begin{frame}{Why a chemical subsystem alone may miss useful structure}
\begin{columns}[T]
\column{.48\textwidth}
\begin{itemize}\setlength\itemsep{3mm}
\item A polymorph is a different crystal structure with the same formula.
\item For Na, the MP20 bank contains several space groups and different energies.
\item A formula-only view groups these structures by chemistry, but it does not describe their structural differences.
\item A candidate may be close to a useful structure even when its best chemical match is outside the exact subsystem.
\end{itemize}

\begin{frame}{Graph model setup}
\begin{columns}[T]
\column{.31\textwidth}
\textbf{Nodes}
\begin{itemize}\setlength\itemsep{2mm}
\item Each node is one known MP20 reference crystal.
\item The node stores its frozen 512-D JEPA embedding.
\item Composition fractions describe the elements in the crystal.
\end{itemize}
\column{.31\textwidth}
\textbf{Edges}
\begin{itemize}\setlength\itemsep{2mm}
\item Main graph: connect references that co-occur in a fit-side hull decomposition.
\item Weight each edge by the pair's co-occurrence count.
\end{itemize}
\column{.31\textwidth}
\textbf{Prediction}
\begin{itemize}\setlength\itemsep{2mm}
\item A generated candidate enters as a query, not as a graph node.
\item Message passing updates each known reference using its neighbours.
\item A candidate-conditioned readout scores reference usefulness; MatterSim evaluates the selected candidates later.
\end{itemize}
\end{columns}
\vspace{3mm}
\centering\small
The graph defines which known references can exchange information. The candidate then receives one final ranking score.
\end{frame}

\begin{frame}{How the reference graph is constructed}
\begin{columns}[T]
\column{.54\textwidth}
\centering
\resizebox{\linewidth}{!}{\begin{tikzpicture}
\node[stage,fill=soft,text width=31mm] (bank) at (0,1.5) {Known reference bank\\27,136 crystals};
\node[stage,fill=soft,text width=31mm] (decomp) at (4.2,1.5) {Fit candidate\\hull decomposition};
\node[stage,fill=soft,text width=31mm] (edge) at (8.4,1.5) {Add reference--reference\\co-occurrence edges};
\draw[arr] (bank)--(decomp);
\draw[arr] (decomp)--(edge);
\node[stage,fill=soft,text width=31mm] (query) at (0,-1.0) {New candidate\\JEPA + composition};
\node[stage,fill=soft,text width=31mm] (pool) at (4.2,-1.0) {Candidate pool\\chemical subset + 5 JEPA kNN};
\node[stage,fill=soft,text width=31mm] (gnn) at (8.4,-1.0) {GCN message passing\\inside the pool};
\draw[arr] (query)--(pool);
\draw[arr] (pool)--(gnn);
\draw[arr] (edge)--(gnn);
\end{tikzpicture}}
\column{.42\textwidth}
\small
\begin{itemize}\setlength\itemsep{3mm}
\item We define one known reference crystal as a node.
\item An edge joins two references that appear together in any candidate's hull decomposition.
\item The edge weight is a function of how often the pair appears together.
\item A generated candidate is used as a query node.
\item At scoring time, the GCN uses only the edges between that candidate's reference pool.
\end{itemize}
\end{columns}
\vspace{3mm}
\centering\small
The graph is built from known references. The candidate selects the local pool and conditions the final reference weights.
\end{frame}

\begin{frame}{Reference selection evaluated with atomistic models}
\begin{itemize}\setlength\itemsep{4mm}
\item Compare \textbf{chemical references}, \textbf{Euclidean/cosine kNN}, and frozen \textbf{GCN, GraphSAGE, and GAT} ranking rules.
\item Evaluate the same held-out candidates with \textbf{MEGNet and CHGNet}, each using its own fixed MP2023 reference hull.
\item The ranking model selects candidates; the atomistic model supplies the proxy energy used to assess them.
\item Question: which rule retrieves more proxy near-stable candidates at the same top-1\% budget?
\end{itemize}
\vspace{3mm}
\centering\color{navy}\textbf{Same candidates and budget; separate, internally consistent proxy hulls.}
\source{Current held-out evaluation: evaluation/evaluation.ipynb. Earlier eSEN results are retained separately as historical evidence.}
\end{frame}

\begin{frame}{Methodology: the held-out 1\% comparison}
\begin{itemize}\setlength\itemsep{2.8mm}
\item \textbf{35,332 reserved candidates}; \textbf{27,136 known MP20 ranking references}. Reuse the original saved JEPA embeddings and frozen GNN weights.
\item Shuffle once into \textbf{58 batches of 600 plus one batch of 532}. Select top \textbf{1\% per batch}, with structural uniqueness and refill: \textbf{353 requested slots per method}.
\item Evaluate the union across methods and budgets: \textbf{10,939 candidate structures} against \textbf{78,066 fixed MP2023 phases}.
\item MEGNet predicts formation energy per atom; CHGNet predicts total energy per atom. Candidates and phases use the \textbf{same model and stored geometries} within each evaluation.
\end{itemize}
\vspace{1mm}
\[
\Delta E_m(c)=E_m(c)-H_m(x_c)<0.1\ \text{eV/atom}.
\]
\centering\small Report the \textbf{best observed run per GNN architecture} and \textbf{best tested k per kNN metric}; preserve unknown labels in the denominator.
\source{Physical/cosine GNNs, three seeds each; kNN k=5,10,15,25. Fixed hulls contain known phases only. Best-run reporting is descriptive and uses evaluation outcomes.}
\end{frame}

\begin{frame}{What the current results mean}
\begin{itemize}\setlength\itemsep{5mm}
\item \textbf{MEGNet:} the best observed GAT run reaches \textbf{81.3\%}, versus \textbf{73.9\%} for chemical references.
\item \textbf{CHGNet:} chemical references reach \textbf{85.6\%} and lead both the best kNN and the best observed GNN runs.
\item \key{No consistent proxy.} No method wins under both proxies --- the observed
ranking improvement depends entirely on which atomistic model supplies the energies.
\end{itemize}

\centering\large\color{navy}\textbf{No consistent proxy signal: the ``best'' ranking rule flips between MEGNet and CHGNet.}
